cmake_minimum_required(VERSION 3.22)
project(tepy_core_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CJSON_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../managed_components/espressif__cjson/cJSON" CACHE PATH "cJSON source (downloaded by IDF)")
if(NOT EXISTS "${CJSON_SOURCE}/cJSON.c")
    message(FATAL_ERROR "Build firmware once to download managed cJSON, or pass -DCJSON_SOURCE=...")
endif()
add_library(cjson STATIC "${CJSON_SOURCE}/cJSON.c")
target_include_directories(cjson PUBLIC "${CJSON_SOURCE}")
target_compile_definitions(cjson PUBLIC CJSON_NESTING_LIMIT=16 CJSON_CIRCULAR_LIMIT=32)
set(IDF_SOURCE "$ENV{IDF_PATH}" CACHE PATH "ESP-IDF source used by the firmware")
if(NOT EXISTS "${IDF_SOURCE}/components/esp_hid/src/esp_hid_common.c")
    message(FATAL_ERROR "Pass -DIDF_SOURCE=<ESP-IDF 6.1 source> to test the actual SDK HID parser")
endif()
add_library(idf_hid_parser STATIC "${IDF_SOURCE}/components/esp_hid/src/esp_hid_common.c")
target_include_directories(idf_hid_parser PRIVATE stubs ../build/config "${IDF_SOURCE}/components/esp_common/include" PUBLIC "${IDF_SOURCE}/components/esp_hid/include")
add_executable(tepy_core_tests core_tests.cpp ../components/tepy_core/core.cpp ../components/tepy_ble/hid_profile.cpp)
target_include_directories(tepy_core_tests PRIVATE ../components/tepy_core/include ../components/tepy_ble/include)
target_link_libraries(tepy_core_tests PRIVATE cjson idf_hid_parser)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(tepy_core_tests PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti)
endif()
enable_testing()
add_test(NAME tepy_core COMMAND tepy_core_tests)
